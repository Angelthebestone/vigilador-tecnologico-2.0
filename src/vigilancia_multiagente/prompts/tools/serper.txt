<tool name="google_search">
  <function_signature>## google_search(query: str, num: int)</function_signature>
  <best_for>general web search using Google results, finding company pages, news articles, documentation</best_for>
  <details>Returns up to 10 organic results with title, URL, snippet. Set num for more (max 20).</details>
  <limits>Timeout: 30s, max results: 20 per call</limits>
  <selection_heuristics>
    <prefer_over>tavily_search, web_search_exa when Google-specific ranking is preferred</prefer_over>
    <avoid_when>You need academic papers (use search_works, google_search_scholar instead)</avoid_when>
  </selection_heuristics>
  <chaining>
    <step order="1">Call google_search with a specific query</step>
    <step order="2">Read promising URLs with read_url or webpage_scrape</step>
    <step order="3">Use the extracted content in your findings</step>
  </chaining>
  <fallback>If result count is low, broaden the query with fewer constraints</fallback>
</tool>

  <chaining>

<tool name="webpage_scrape">
  <function_signature>## webpage_scrape(url: str)</function_signature>
  <best_for>scraping and extracting readable content from a web page URL</best_for>
  <details>Downloads a page and returns its text content. Handles most URL formats.</details>
  <limits>Timeout: 30s</limits>
  <selection_heuristics>
    <prefer_over>read_url (jina) when Serper's extraction is sufficient</prefer_over>
    <avoid_when>The page requires JavaScript rendering (use browser_navigate instead)</avoid_when>
  </selection_heuristics>
  <chaining>
    <step order="1">Search with google_search to find relevant pages</step>
    <step order="2">Extract detailed content with webpage_scrape</step>
    <step order="3">Use extracted data in findings or pass to execute_code for analysis</step>
  </chaining>
</tool>
