<tool name="search_google_scholar_key_words">
  <function_signature>## search_google_scholar_key_words(query: str, num_results: 5)</function_signature>
  <best_for>academic paper search by keywords</best_for>
  <pricing>No API key required</pricing>
  <details>Slow (scrapes Google Scholar). May be blocked by intensive use.</details>
  <recommendation>Keep num_results low (5-10). Use sparingly.</recommendation>
  <limits>Timeout: 25s, retries: 2</limits>
  <selection_heuristics>
    <use_when>You need academic papers and ArXiv doesn't cover the topic</use_when>
    <use_when>You want citation and impact metrics (citations per paper)</use_when>
    <prefer_over>ArXiv for social sciences, medicine, humanities papers</prefer_over>
    <avoid_when>ArXiv has good coverage of the topic (CS, physics, math)</avoid_when>
    <avoid_when>You need the full paper text (Scholar only gives metadata)</avoid_when>
  </selection_heuristics>
  <chaining>
    <step order="1">search_google_scholar_key_words to find relevant papers</step>
    <step order="2">Extract title/author/year and search in ArXiv or Jina read_url for full text</step>
  </chaining>
  <fallback>If it fails (blocked by scraping), use search_papers from ArXiv with same query</fallback>
</tool>

<tool name="get_author_info">

  <best_for>profiling a specific researcher: affiliation, citation metrics, research areas</best_for>
  <details>Returns author-level data (not papers). Use to map key players and expertise of competitors.</details>
  <limits>Timeout: 25s, retries: 2</limits>
  <selection_heuristics>
    <use_when>You identified a relevant author and need their full profile / impact</use_when>
    <use_when>Competitive intelligence: who leads research at a given organization</use_when>
    <prefer_over>OpenAlex get_author_profile when the name comes from a Scholar result</prefer_over>
    <avoid_when>You need a list of papers (use the search tools instead)</avoid_when>
  </selection_heuristics>
  <fallback>If it fails (scraping blocked), use OpenAlex search_authors_by_expertise with the topic</fallback>
</tool>
