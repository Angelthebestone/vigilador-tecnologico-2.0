<tool name="brave_web_search">
  <function_signature>## brave_web_search(query: str, count: 1-20, offset: 0-9, freshness: "pd"|"pw"|"pm"|"py"|"YYYY-MM-DDtoYYYY-MM-DD", country: str, search_lang: str, result_filter: str[])</function_signature>
  <best_for>general web search, privacy-focused</best_for>
  <constraints>Max query length: 400 chars, 50 words</constraints>
  <parameters>
    <param name="freshness" values="pd|pw|pm|py|date-range" description="pd=past day, pw=week, pm=month, py=year; also accepts an explicit YYYY-MM-DD to YYYY-MM-DD range"/>
    <param name="country" description="2-letter code (default US) — use for market-specific intelligence (e.g. MX, IN, DE)"/>
    <param name="search_lang" description="search language (default en) — pair with country for localized results"/>
    <param name="offset" description="pagination offset 0-9 — go beyond the first page for broader coverage"/>
    <param name="result_filter" description="restrict result types, e.g. ['web','news']"/>
  </parameters>
  <limits>Timeout: 20s, retries: 2</limits>
  <selection_heuristics>
    <prefer_over>Tavily when query is ≤400 chars and you want fresh links</prefer_over>
    <prefer_over>Exa when you need high-privacy results</prefer_over>
    <avoid_when>Query exceeds 400 chars (use Tavily or Exa)</avoid_when>
    <avoid_when>You need synthesized answers (Tavily include_answer is better)</avoid_when>
  </selection_heuristics>
  <chaining>
    <step order="1">brave_web_search to find recent URLs</step>
    <step order="2">Jina read_url or fetch to extract content from URLs</step>
  </chaining>
  <fallback>If it fails, retry with count=5 and freshness="pw" (week)</fallback>
</tool>

<tool name="brave_news_search">
  <function_signature>## brave_news_search(query: str, count: 1-50, offset: 0-9, freshness: "pd"|"pw"|"pm"|"py", country: str, search_lang: str)</function_signature>
  <best_for>real-time news search</best_for>
  <parameters>
    <param name="freshness" default="pd" description="last 24 hours; pw/pm/py to widen"/>
    <param name="country" description="2-letter code — localize news to a target market"/>
    <param name="offset" description="pagination 0-9 for deeper news coverage"/>
  </parameters>
  <limits>Timeout: 15s, retries: 2</limits>
  <selection_heuristics>
    <use_when>You need breaking news (last 24-48h)</use_when>
    <use_when>The topic has high volatility (launches, breaches, regulations)</use_when>
    <prefer_over>Google News via Exa for English-language news</prefer_over>
  </selection_heuristics>
  <fallback>If it fails, use web_search_exa with startPublishedDate=recent_date</fallback>
</tool>

<rules>
  <rule>Brave is best for real-time news. Use freshness="pd" by default.</rule>
  <rule>For market-specific intelligence set country/search_lang (e.g. country="MX" for Mexican market signals).</rule>
  <rule>Use a freshness date-range (YYYY-MM-DD to YYYY-MM-DD) when the analysis window is fixed; use offset to go past the first page.</rule>
  <rule>For deep technical content, prefer Exa or Tavily with search_depth="advanced".</rule>
  <rule>After finding URLs, extract content with Jina read_url or fetch.</rule>
</rules>
